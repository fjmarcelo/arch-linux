\section{Entornos gráficos}
\subsection{Instalación}

Como ya podemos usar {\color{brown}\texttt{yay}}

  \begin{tcolorbox}[title=Para instalar \textbf{GNOME}]
    \orden{yay -S gnome gdm}

    \orden{systemctl enable gdm}
  \end{tcolorbox}

  \begin{tcolorbox}[title=Para instalar \textbf{Plasma}]
    \orden{yay -S plasma sddm}

    \orden{systemctl enable sddm}
  \end{tcolorbox}

    {\color{blue}Podemos instalar cualquier escritorio de esta manera tan sencilla.}
