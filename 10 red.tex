\section{Configuración de la red}
\subsection{Nombre del equipo (\textit{hostname})}
\begin{tcolorbox}[title=Le damos nombre a nuestro ordenador]
\begin{lstlisting}[style=bashstyle]
echo Lusitania > /etc/hostname
\end{lstlisting}
\end{tcolorbox}

\subsection{Fichero \texttt{/etc/hosts}}

\begin{tcolorbox}[title=Añadimos las siguientes líneas en {\ttfamily /etc/hosts}]
\begin{lstlisting}[style=bashstyle]
127.0.0.1  localhost
::1        localhost
127.0.1.1  Lusitania
\end{lstlisting}
\end{tcolorbox}
    

\begin{tcolorbox}[title=Habilitamos {\ttfamily NetworkManager} para su arranque automático]
\begin{lstlisting}[style=bashstyle]
systemctl enable NetworkManager
\end{lstlisting}
\end{tcolorbox}
