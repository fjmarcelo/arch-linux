\section{Comprobamos qué tipo de arranque tiene la placa}
\begin{frame}

\begin{center}
\color{blue}\color{black}\vspace{5mm}
    
    
    \begin{tcolorbox}[title=Comprobamos qué tipo de arranque tiene la placa]
	     \orden{ls /sys/firmware/efi/efivars}
    \end{tcolorbox}\pause
    
    \only<1>{\includegraphics[width=80mm]{3 UEFI.png}\\}
    \only<1>{Si vemos esta salida, entonces es un arranque \textbf{\color{red}UEFI}.}
    
    \only<1>{Eso significa que deberemos instalar los paquetes {\ttfamily \color{blue}grub} y {\ttfamily \color{blue}efibootmgr}.}
    
    \only<2>{Si no vemos nada tras la ejecución, entonces es un arranque \textbf{\color{red}BIOS}.}
    
    \only<2>{En este caso deberemos instalar el paquete {\ttfamily \color{blue}grub-bios}.}
    
    \end{center}
\end{frame}
