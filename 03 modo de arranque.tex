 \end{tcolorbox}
 
 \includegraphics[width=.95\textwidth]{3 UEFI.png}\\

 \vspace{15mm}

 \begin{tcolorbox}[title=Si  vemos  esta  salida\,  entonces  es  un  arranque  \btrojo{UEFI}]
Eso  significa  que  deberemos  instalar  los paquetes  \tazul{grub}  y  \tazul{efibootmgr}.
 \end{tcolorbox}

 \vspace{15mm}

 \begin{tcolorbox}[title=Si  no la vemos entonces  es  un arranque  \brojo{BIOS}]
En  este  caso  deberemos instalar  el  paquete   \tazul{grub-bios}.
 \end{tcolorbox}
