\section{Instalación de {\color{brown}\texttt{yay}}}
\subsection{Proceso de instalación}

\warning{\texttt{yay} no puede ser ejecutado por \textbf{root}. Los siguientes pasos deben realizarse como el usuario \textbf{viriato}. Cambia de usuario antes de continuar:
\begin{lstlisting}[style=bashstyle]
su - viriato
\end{lstlisting}}

\vspace{3mm}
	\begin{center}
		{\color{blue}Pasos a seguir:}\vspace{3mm}

      \begin{tcolorbox}[title=Descargamos los ficheros desde el repositorio \textbf{AUR}:]
			\orden{git clone https://aur.archlinux.org/yay.git}
      \end{tcolorbox}

      \begin{tcolorbox}[title=Ahora vamos al directorio y compilamos:]
			\orden{cd yay}\vspace{1mm}

			\orden{makepkg -si}
      \end{tcolorbox}

		Para instalar software solo hay que escribir {\color{red}\texttt{yay}} en lugar de {\color{red}\texttt{pacman}} sin necesidad de usar la orden {\color{red}\texttt{sudo}}.\\[3mm]
		Desde este momento podemos instalar todos los paquetes con {\color{red}\texttt{yay}}.
		\end{center}
